        <div class="version-chain">
          <div *ngFor="let item of versionChain; let i = index"

                 'latest': item.isLatest,
                 'ellipsis': item.type === 'ellipsis',
                 'gap': item.type === 'gap',
                 'future': item.type === 'future'
               }">
            
            <!-- Ellipsis -->
            <div *ngIf="item.type === 'ellipsis'" class="version-content ellipsis-content">

            </div>

            <div *ngIf="item.type === 'gap'" class="version-content gap-content">
              <span class="version-text">{{ item.version }}</span>
            </div>
            
            <!-- Current version -->
            <div *ngIf="item.type === 'current'" class="version-content current-content" (click)="selectVersionInfo(item.version)">
              
              <!-- Development version: version number is not clickable -->
              <span class="version-text" *ngIf="versionStatus === 'ahead'">v{{ item.version }}</span>
              
              <!-- Standard version: version number is clickable -->
              <span class="version-text clickable-version" *ngIf="versionStatus !== 'ahead'" (click)="downloadVersionFiles(item.version); $event.stopPropagation()">v{{ item.version }}</span>
              
              <!-- Display when the current version is a development version (ahead of the release) -->
              <small class="version-label" *ngIf="versionStatus === 'ahead'">
                You are here now<br>
                Under develop
              </small>
              
              <!-- Display when the current version is a standard version -->
              <small class="version-label" *ngIf="versionStatus !== 'ahead'">
                You are here now
              </small>
              
              <div class="current-indicator" [class.dev-version-indicator]="versionStatus === 'ahead'">
                <i class="pi pi-map-marker"></i>
              </div>
            </div>
            
            <!-- Release version -->
            <div *ngIf="item.type === 'release'" 
                 class="version-content release-content"
                 [class.latest-release]="item.isLatest"
                 [class.current-content]="item.isCurrent"
                 (click)="selectVersionInfo(item.version)">
              <span class="version-text clickable-version" (click)="downloadVersionFiles(item.version); $event.stopPropagation()">v{{ item.version }}</span>
              
              <!-- Both the latest version and the current version -->
              <small class="version-label" *ngIf="item.isLatest && item.isCurrent">
                The Latest<br>
                You are here now
              </small>
              
              <!-- Latest version only (not the current version) -->
              <small class="version-label" *ngIf="item.isLatest && !item.isCurrent">
                Latest Release<br>
                <span class="release-date">{{ item.publishedAt | date:'MMM d, y' }}</span>
              </small>
              
              <!-- Current version only (not the latest version) -->
              <small class="version-label" *ngIf="!item.isLatest && item.isCurrent">
                You are here now
              </small>
              
              <!-- Neither the latest version nor the current version -->
              <small class="version-label" *ngIf="!item.isLatest && !item.isCurrent">
                {{ item.publishedAt | date:'MMM d, y' }}
              </small>
              <!-- Indicator for both the latest version and the current version -->
              <div *ngIf="item.isLatest && item.isCurrent" class="latest-current-indicator">
                <i class="pi pi-star"></i>
                <i class="pi pi-map-marker"></i>
              </div>
              
              <!-- Indicator for the latest version only -->
              <div *ngIf="item.isLatest && !item.isCurrent" class="latest-indicator">
                <i class="pi pi-star"></i>
              </div>
              
              <!-- Indicator for the current version only -->
              <div *ngIf="!item.isLatest && item.isCurrent" class="current-indicator">
                <i class="pi pi-map-marker"></i>
              </div>
            </div>
            
            <!-- Future version -->
            <div *ngIf="item.type === 'future'" class="version-content future-content">
              <span class="version-text">{{ item.version }}</span>
              <small class="version-label">Future</small>
            </div>
            
            <!-- Connecting line -->
            <div *ngIf="i < versionChain.length - 1" class="version-connector">
              <div class="connector-line"></div>
              <div class="connector-arrow">
                <i class="pi pi-caret-right"></i>
              </div>
            </div>
          </div>
        </div>
      </div>
    </div>
  </div>
</div>
